-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- PROFILES TABLE
create table profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  username text unique not null,
  full_name text,
  bio text,
  avatar_url text,
  theme text default 'default',
  github_username text,
  created_at timestamp with time zone default now()
);

-- LINKS TABLE
create table links (
  id uuid primary key default uuid_generate_v4(),
  user_id uuid references auth.users(id) on delete cascade,
  title text not null,
  url text not null,
  icon text,
  clicks integer default 0,
  position integer default 0,
  created_at timestamp with time zone default now()
);

-- ANALYTICS TABLE
create table analytics (
  id uuid primary key default uuid_generate_v4(),
  profile_id uuid references profiles(id) on delete cascade,
  event_type text not null, -- 'view' or 'click'
  link_id uuid references links(id) on delete set null,
  ip_hash text,
  created_at timestamp with time zone default now()
);

-- ROW LEVEL SECURITY
alter table profiles enable row level security;
alter table links enable row level security;
alter table analytics enable row level security;

-- Profiles policies
create policy "Public profiles are viewable by everyone"
  on profiles for select using (true);

create policy "Users can insert own profile"
  on profiles for insert with check (auth.uid() = id);

create policy "Users can update own profile"
  on profiles for update using (auth.uid() = id);

-- Links policies
create policy "Links are viewable by everyone"
  on links for select using (true);

create policy "Users can manage own links"
  on links for all using (auth.uid() = user_id);

-- Analytics policies
create policy "Anyone can insert analytics"
  on analytics for insert with check (true);

create policy "Users can view own analytics"
  on analytics for select using (
    profile_id in (select id from profiles where id = auth.uid())
  );

const supabase = require('../lib/supabaseAdmin');

const authenticate = async (req, res, next) => {
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Missing or invalid token' });
  }

  const token = authHeader.split(' ')[1];

  const { data: { user }, error } = await supabase.auth.getUser(token);
  if (error || !user) {
    return res.status(401).json({ error: 'Unauthorized' });
  }

  req.user = user;
  next();
};

module.exports = authenticate;

router.get('/:username', async (req, res) => {

  const { data: profile, error } = await supabase
    .from('profiles')
    .select('*, links(*)')
    .eq('username', username)

  if (error || !profile) {
    return res.status(404).json({ error: 'Profile not found' });

  // Sort links by position

router.post('/', authenticate, async (req, res) => {
  const { username, full_name, bio, theme, github_username, avatar_url } = req.body;

  if (!username || !/^[a-zA-Z0-9_-]{3,30}$/.test(username)) {
    return res.status(400).json({ error: 'Invalid username (3-30 chars, letters/numbers/_/-)' });

  const { data, error } = await supabase
    .from('profiles')
    .insert([{ id: req.user.id, username, full_name, bio, theme, github_username, avatar_url }])

    .single();

  if (error) {
    if (error.code === '23505') return res.status(409).json({ error: 'Username already taken' });

  }

  res.status(201).json(data);
});

const router = express.Router();
const supabase = require('../lib/supabaseAdmin');

const FREE_LINK_LIMIT = 5;

// GET all links for current user
router.get('/', authenticate, async (req, res) => {
  const { data, error } = await supabase
    .from('links')
    .select('*')
    .eq('user_id', req.user.id)
    .order('position', { ascending: true });

  if (error) return res.status(500).json({ error: error.message });
  res.json(data);
});

// POST create link
router.post('/', authenticate, async (req, res) => {
  const { title, url, icon } = req.body;

  // Check free tier limit
  const { count } = await supabase
    .from('links')
    .select('*', { count: 'exact', head: true })
    .eq('user_id', req.user.id);

  if (count >= FREE_LINK_LIMIT) {
    return res.status(403).json({ error: 'Free plan limited to 5 links. Upgrade for unlimited.' });
  }

  const { data, error } = await supabase
    .from('links')
    .insert([{ user_id: req.user.id, title, url, icon, position: count }])
    .select()
    .single();

  if (error) return res.status(500).json({ error: error.message });
  res.status(201).json(data);
});

// PUT update link
router.put('/:id', authenticate, async (req, res) => {
  const { title, url, icon, position } = req.body;

  const { data, error } = await supabase
    .from('links')
    .update({ title, url, icon, position })
    .eq('id', req.params.id)
    .eq('user_id', req.user.id)
    .select()
    .single();

  if (error) return res.status(500).json({ error: error.message });
  res.json(data);
});

// DELETE link
router.delete('/:id', authenticate, async (req, res) => {
  const { error } = await supabase
    .from('links')
    .delete()
    .eq('id', req.params.id)
    .eq('user_id', req.user.id);

  if (error) return res.status(500).json({ error: error.message });
  res.json({ message: 'Link deleted' });
});

// POST track click (public)
router.post('/:id/click', async (req, res) => {
  const { data: link } = await supabase
    .from('links')
    .select('id, user_id, clicks')
    .eq('id', req.params.id)
    .single();

  if (!link) return res.status(404).json({ error: 'Link not found' });

  await supabase.from('links').update({ clicks: link.clicks + 1 }).eq('id', link.id);

  await supabase.from('analytics').insert([{
    profile_id: link.user_id,
    event_type: 'click',
    link_id: link.id,
  }]);

  res.json({ success: true });
});

module.exports = router;

const express = require('express');
const router = express.Router();
const supabase = require('../lib/supabaseAdmin');
const authenticate = require('../middleware/auth');

// POST track profile view (public)
router.post('/view/:username', async (req, res) => {
  const { data: profile } = await supabase
    .from('profiles')
    .select('id')
    .eq('username', req.params.username)
    .single();

  if (!profile) return res.status(404).json({ error: 'Profile not found' });

  await supabase.from('analytics').insert([{
    profile_id: profile.id,
    event_type: 'view',
  }]);

  res.json({ success: true });
});

// GET analytics summary for current user
router.get('/summary', authenticate, async (req, res) => {
  const { data: profile } = await supabase
    .from('profiles')
    .select('id')
    .eq('id', req.user.id)
    .single();

  if (!profile) return res.status(404).json({ error: 'Profile not found' });

  const { data: views } = await supabase
    .from('analytics')
    .select('created_at')
    .eq('profile_id', profile.id)
    .eq('event_type', 'view');

  const { data: clicks } = await supabase
    .from('analytics')
    .select('created_at, link_id')
    .eq('profile_id', profile.id)
    .eq('event_type', 'click');

  // Group views by date (last 7 days)
  const now = new Date();
  const days = Array.from({ length: 7 }, (_, i) => {
    const d = new Date(now);
    d.setDate(d.getDate() - (6 - i));
    return d.toISOString().split('T')[0];
  });

  const viewsByDay = days.map(day => ({
    date: day,
    views: views.filter(v => v.created_at.startsWith(day)).length,
  }));

  res.json({
    total_views: views.length,
    total_clicks: clicks.length,
    views_by_day: viewsByDay,
  });
});

      setUser(session?.user ?? null)
      setLoading(false)
    })

    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      setUser(session?.user ?? null)

  }, [])

    supabase.auth.signUp({ email, password })

  const signIn = (email, password) =>
    supabase.auth.signInWithPassword({ email, password })

  const signOut = () => supabase.auth.signOut()

  return (
    <AuthContext.Provider value={{ user, loading, signUp, signIn, signOut }}>
      {children}
    </AuthContext.Provider>
  )
}

import { useState } from 'react'
import { useNavigate, Link } from 'react-router-dom'
import { useAuth } from '../context/AuthContext'
import api from '../lib/api'

export default function Register() {
  const { signUp } = useAuth()
  const navigate = useNavigate()
  const [form, setForm] = useState({ email: '', password: '', username: '' })
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  const handleSubmit = async (e) => {
    e.preventDefault()
    setError('')
    setLoading(true)

    const { error: authError } = await signUp(form.email, form.password)
    if (authError) { setLoading(false); return setError(authError.message) }

    // Wait a tick for auth state to settle, then create profile
    setTimeout(async () => {
      try {
        await api.post('/profile', { username: form.username })
        navigate('/dashboard')
      } catch (err) {
        setError(err.response?.data?.error || 'Profile creation failed')
      }
      setLoading(false)
    }, 1500)
  }

  return (
    <div className="min-h-screen flex items-center justify-center px-4">
      <div className="card w-full max-w-sm">
        <h1 className="text-xl font-semibold text-white mb-6">Create your DevLink</h1>
        {error && <p className="text-red-400 text-sm mb-4">{error}</p>}
        <form onSubmit={handleSubmit} className="space-y-4">

import { useNavigate } from 'react-router-dom'
import api from '../lib/api'
import Navbar from '../components/Navbar'
import ThemePicker from '../components/ThemePicker'

export default function EditProfile() {
  const navigate = useNavigate()
  const [profile, setProfile] = useState(null)
  const [links, setLinks] = useState([])
  const [newLink, setNewLink] = useState({ title: '', url: '' })
  const [saving, setSaving] = useState(false)
  const [msg, setMsg] = useState('')

  useEffect(() => {
    api.get('/profile').then(r => {
      setProfile(r.data)
      setLinks(r.data.links || [])
    })
  }, [])

  const saveProfile = async () => {
    setSaving(true)
    await api.put('/profile', {
      full_name: profile.full_name,
      bio: profile.bio,
      theme: profile.theme,
      github_username: profile.github_username,
    })
    setMsg('Saved!')
    setSaving(false)
    setTimeout(() => setMsg(''), 2000)
  }

  const addLink = async () => {
    if (!newLink.title || !newLink.url) return
    try {
      const res = await api.post('/links', newLink)
      setLinks([...links, res.data])
      setNewLink({ title: '', url: '' })
    } catch (err) {
      alert(err.response?.data?.error || 'Failed to add link')
    }
  }

  const deleteLink = async (id) => {
    await api.delete(`/links/${id}`)
    setLinks(links.filter(l => l.id !== id))
  }

  if (!profile) return <div className="min-h-screen flex items-center justify-center text-gray-400">Loading...</div>

  return (
    <div className="min-h-screen bg-gray-950">
      <Navbar username={profile.username} />
      <main className="max-w-2xl mx-auto px-4 py-10 space-y-8">
        <h1 className="text-2xl font-bold text-white">Edit Profile</h1>

        {/* Profile info */}
        <div className="card space-y-4">
          <h2 className="text-white font-semibold">Profile info</h2>
          <input className="input-field" placeholder="Full name" value={profile.full_name || ''} onChange={e => setProfile({...profile, full_name: e.target.value})} />
          <textarea className="input-field resize-none h-24" placeholder="Bio" value={profile.bio || ''} onChange={e => setProfile({...profile, bio: e.target.value})} />
          <input className="input-field" placeholder="GitHub username" value={profile.github_username || ''} onChange={e => setProfile({...profile, github_username: e.target.value})} />
        </div>

        {/* Theme */}
        <div className="card">
          <h2 className="text-white font-semibold mb-4">Theme</h2>
          <ThemePicker selected={profile.theme} onChange={theme => setProfile({...profile, theme})} />
        </div>

        <div className="flex items-center gap-4">
          <button className="btn-primary" onClick={saveProfile} disabled={saving}>{saving ? 'Saving...' : 'Save profile'}</button>
          {msg && <span className="text-green-400 text-sm">{msg}</span>}
        </div>

        {/* Links */}
        <div className="card space-y-4">
          <h2 className="text-white font-semibold">Links ({links.length}/5 on free plan)</h2>
          {links.map(link => (
            <div key={link.id} className="flex items-center gap-3">
              <div className="flex-1">
                <p className="text-white text-sm font-medium">{link.title}</p>
                <p className="text-gray-500 text-xs font-mono truncate">{link.url}</p>
              </div>
              <button onClick={() => deleteLink(link.id)} className="text-red-500 hover:text-red-400 text-sm">Delete</button>
            </div>
          ))}
          <div className="border-t border-gray-800 pt-4 space-y-3">
            <p className="text-gray-400 text-sm font-semibold">Add a link</p>
            <input className="input-field" placeholder="Title (e.g. GitHub)" value={newLink.title} onChange={e => setNewLink({...newLink, title: e.target.value})} />
            <input className="input-field" placeholder="URL (https://...)" value={newLink.url} onChange={e => setNewLink({...newLink, url: e.target.value})} />
            <button className="btn-primary" onClick={addLink}>Add link</button>
          </div>
        </div>

        <button className="btn-outline" onClick={() => navigate('/dashboard')}>← Back to dashboard</button>
      </main>
    </div>
  )
}

   In production, set VITE_API_BASE_URL to your Render URL and update api.js:
   baseURL: import.meta.env.VITE_API_BASE_URL || '/api'
