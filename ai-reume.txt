CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Users (mirrors Supabase Auth)
CREATE TABLE users (
  id         UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  email      TEXT UNIQUE NOT NULL,
  plan       TEXT NOT NULL DEFAULT 'free',    -- 'free' | 'paid'
  daily_used INTEGER NOT NULL DEFAULT 0,
  last_reset DATE NOT NULL DEFAULT CURRENT_DATE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Resumes
CREATE TABLE resumes (
  id          UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id     UUID REFERENCES users(id) ON DELETE CASCADE,
  title       TEXT NOT NULL DEFAULT 'Untitled Resume',
  full_name   TEXT,
  email       TEXT,
  phone       TEXT,
  location    TEXT,
  skills      TEXT,
  experience  TEXT,
  job_title   TEXT,
  job_desc    TEXT,
  content     TEXT,           -- AI-generated resume text
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- Cover Letters
CREATE TABLE cover_letters (
  id          UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id     UUID REFERENCES users(id) ON DELETE CASCADE,
  resume_id   UUID REFERENCES resumes(id) ON DELETE SET NULL,
  title       TEXT NOT NULL DEFAULT 'Untitled Cover Letter',
  content     TEXT,           -- AI-generated cover letter text
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- Generations log (for cost tracking & caching)
CREATE TABLE generations (
  id          UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id     UUID REFERENCES users(id) ON DELETE CASCADE,
  type        TEXT NOT NULL,  -- 'resume' | 'cover_letter'
  prompt_hash TEXT,           -- SHA256 of prompt for cache lookup
  prompt      TEXT,
  response    TEXT,
  tokens_used INTEGER DEFAULT 0,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- Row Level Security
ALTER TABLE users         ENABLE ROW LEVEL SECURITY;
ALTER TABLE resumes       ENABLE ROW LEVEL SECURITY;
ALTER TABLE cover_letters ENABLE ROW LEVEL SECURITY;
ALTER TABLE generations   ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users own their profile"
  ON users FOR ALL USING (auth.uid() = id);

CREATE POLICY "Users own their resumes"
  ON resumes FOR ALL USING (auth.uid() = user_id);

CREATE POLICY "Users own their cover letters"
  ON cover_letters FOR ALL USING (auth.uid() = user_id);

CREATE POLICY "Users own their generations"
  ON generations FOR ALL USING (auth.uid() = user_id);

module.exports = async function authMiddleware(req, res, next) {
  const authHeader = req.headers.authorization;
  if (!authHeader) return res.status(401).json({ error: 'No token provided' });

  const token = authHeader.replace('Bearer ', '');
  const { data, error } = await supabase.auth.getUser(token);

  if (error || !data.user) {
    return res.status(401).json({ error: 'Invalid or expired token' });
  }

  req.user = data.user;
  next();
};

async function generateResume({ fullName, email, phone, location, skills, experience, jobTitle, jobDesc }) {
  const prompt = `Generate a professional, ATS-friendly resume for the following person.

Personal Details:
- Name: ${fullName}
- Email: ${email}
- Phone: ${phone || 'N/A'}
- Location: ${location || 'N/A'}

Skills: ${skills}
Experience: ${experience}
Target Job Title: ${jobTitle}
Job Description: ${jobDesc}

Format the resume with these clearly labeled sections:
## SUMMARY
(2-3 sentence professional summary tailored to the job)

## SKILLS
(bullet list of relevant skills)

## EXPERIENCE
(formatted work experience entries)

## EDUCATION
(placeholder if not provided)

## PROJECTS
(suggest 1-2 relevant project ideas based on skills)

Keep the tone professional. Tailor the content specifically to the job description provided.`;

  const completion = await openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: prompt }],
    max_tokens: 1200,
    temperature: 0.7,
  });

  return {
    content: completion.choices[0].message.content,
    tokens: completion.usage.total_tokens,
  };
}

async function generateCoverLetter({ fullName, skills, experience, jobTitle, jobDesc, companyName }) {
  const prompt = `Write a compelling, professional cover letter for a job application.

Applicant: ${fullName}
Skills: ${skills}
Experience: ${experience}
Target Role: ${jobTitle}
Company: ${companyName || 'the company'}
Job Description: ${jobDesc}

Structure:
- Opening paragraph: Express enthusiasm for the role and company
- Middle paragraph: Highlight 2-3 most relevant skills/experiences
- Closing paragraph: Call to action and thank you

Keep it to 3 paragraphs, confident and personalized. Do not use generic filler phrases.`;

  const completion = await openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: prompt }],
    max_tokens: 600,
    temperature: 0.75,
  });

  return {
    content: completion.choices[0].message.content,
    tokens: completion.usage.total_tokens,
  };
}

module.exports = { generateResume, generateCoverLetter };

const PDFDocument = require('pdfkit');

function generateResumePDF(resume, stream) {
  const doc = new PDFDocument({ margin: 55, size: 'A4' });
  doc.pipe(stream);

  // Header
  doc.fontSize(22).font('Helvetica-Bold')
     .text(resume.full_name || 'Resume', { align: 'center' });

  const contactParts = [resume.email, resume.phone, resume.location].filter(Boolean);
  if (contactParts.length) {
    doc.moveDown(0.3).fontSize(9).font('Helvetica')
       .fillColor('#555555')
       .text(contactParts.join('  |  '), { align: 'center' });
  }

  doc.moveDown(0.5)
     .moveTo(55, doc.y).lineTo(540, doc.y)
     .lineWidth(1).strokeColor('#4f46e5').stroke();

  doc.moveDown(0.8).fillColor('#000000');

  // Parse and render sections from AI content
  const sections = (resume.content || '').split(/\n## /);
  sections.forEach((section, idx) => {
    const lines = section.trim().split('\n');
    const heading = idx === 0 ? null : lines[0].replace('## ', '').trim();
    const body    = idx === 0 ? lines.join('\n') : lines.slice(1).join('\n');

    if (heading) {
      doc.fontSize(11).font('Helvetica-Bold')
         .fillColor('#4f46e5')
         .text(heading.toUpperCase(), { continued: false });
      doc.moveTo(55, doc.y + 2).lineTo(540, doc.y + 2)
         .lineWidth(0.5).strokeColor('#c7d2fe').stroke();
      doc.moveDown(0.4);
    }

    doc.fontSize(9.5).font('Helvetica').fillColor('#111111')
       .text(body.trim(), { lineGap: 3 });
    doc.moveDown(0.8);
  });

  doc.end();
}

function generateCoverLetterPDF(coverLetter, resume, stream) {
  const doc = new PDFDocument({ margin: 55, size: 'A4' });
  doc.pipe(stream);

  // Header
  doc.fontSize(20).font('Helvetica-Bold')
     .text('Cover Letter', { align: 'center' });

  if (resume?.full_name) {
    doc.moveDown(0.3).fontSize(10).font('Helvetica')
       .fillColor('#555555')
       .text(resume.full_name, { align: 'center' });
  }

  doc.moveDown(0.5)
     .moveTo(55, doc.y).lineTo(540, doc.y)
     .lineWidth(1).strokeColor('#4f46e5').stroke();

  doc.moveDown(1).fillColor('#111111').fontSize(10.5).font('Helvetica')
     .text(coverLetter.content || '', { lineGap: 5 });

  doc.end();
}

module.exports = { generateResumePDF, generateCoverLetterPDF };

const router   = express.Router();

const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY
);

const DAILY_FREE_LIMIT = parseInt(process.env.DAILY_FREE_LIMIT || '5', 10);

// Helper: enforce daily usage limit
async function checkAndIncrementUsage(userId) {
  const today = new Date().toISOString().split('T')[0];

  let { data: user } = await supabase
    .from('users')
    .select('plan, daily_used, last_reset')
    .eq('id', userId)
    .single();

  // Upsert user row if it doesn't exist yet
  if (!user) {
    await supabase.from('users').upsert({ id: userId, email: '', daily_used: 0, last_reset: today });
    user = { plan: 'free', daily_used: 0, last_reset: today };
  }

  // Reset counter if it's a new day
  if (user.last_reset !== today) {
    await supabase.from('users')
      .update({ daily_used: 0, last_reset: today })
      .eq('id', userId);
    user.daily_used = 0;
  }

  if (user.plan === 'free' && user.daily_used >= DAILY_FREE_LIMIT) {
    return { allowed: false, used: user.daily_used, limit: DAILY_FREE_LIMIT };
  }

  await supabase.from('users')
    .update({ daily_used: user.daily_used + 1 })
    .eq('id', userId);

  return { allowed: true, used: user.daily_used + 1, limit: DAILY_FREE_LIMIT, plan: user.plan };
}

// POST /api/generate/resume
router.post('/resume', auth, generateLimiter, async (req, res) => {
  const usage = await checkAndIncrementUsage(req.user.id);
  if (!usage.allowed) {
    return res.status(403).json({
      error: `Daily limit of ${usage.limit} generations reached. Upgrade to Pro for unlimited access.`,
      usage,
    });

      fullName, email, phone, location, skills, experience, jobTitle, jobDesc,
    });

      .from('resumes')

        user_id: req.user.id,

        full_name: fullName,
        email,
        phone,
        location,
        skills,
        experience,
        job_title: jobTitle,
        job_desc: jobDesc,
        content,
      })
      .select()
      .single();

    if (error) return res.status(500).json({ error: error.message });

    // Log generation
    await supabase.from('generations').insert({
      user_id: req.user.id,
      type: 'resume',
      prompt: `${jobTitle} | ${skills}`,
      response: content,
      tokens_used: tokens,
    });

export async function apiFetch(path, options = {}) {
  const { data: { session } } = await supabase.auth.getSession();
  const token = session?.access_token;

  const res = await fetch(`/api${path}`, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${token}`,
      ...(options.headers || {}),
    },
  });

  if (!res.ok) {
    const err = await res.json().catch(() => ({ error: res.statusText }));
    throw new Error(err.error || 'Request failed');
  }

  // PDF responses should be handled by caller
  if (res.headers.get('Content-Type')?.includes('application/pdf')) {
    return res.blob();
  }

  return res.json();
}

export async function downloadPDF(path, filename) {
  const { data: { session } } = await supabase.auth.getSession();
  const token = session?.access_token;

  const res = await fetch(`/api${path}`, {
    headers: { Authorization: `Bearer ${token}` },
  });

  if (!res.ok) throw new Error('PDF download failed');

  const blob = await res.blob();
  const url  = URL.createObjectURL(blob);
  const a    = document.createElement('a');
  a.href     = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}

import { createContext, useContext, useEffect, useState } from 'react';
import { supabase } from '../api';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser]       = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => {
      setUser(data.session?.user ?? null);
      setLoading(false);
    });

    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      (_event, session) => setUser(session?.user ?? null)
    );

    return () => subscription.unsubscribe();
  }, []);

  const signIn  = (email, password) => supabase.auth.signInWithPassword({ email, password });
  const signUp  = (email, password) => supabase.auth.signUp({ email, password });
  const signOut = () => supabase.auth.signOut();

  return (
    <AuthContext.Provider value={{ user, loading, signIn, signUp, signOut }}>
      {children}
    </AuthContext.Provider>
  );
}

export const useAuth = () => useContext(AuthContext);

      <div className="w-8 h-8 border-4 border-brand-500 border-t-transparent rounded-full animate-spin" />
    </div>
  );
  return user ? children : <Navigate to="/login" replace />;
}

function Layout({ children }) {
  return (
    <>
      <Navbar />
      <main className="max-w-5xl mx-auto px-4 py-8">{children}</main>
    </>
  );
}

export default function App() {
  return (
    <AuthProvider>
      <BrowserRouter>
        <Routes>
          <Route path="/login" element={<Login />} />
          <Route path="/" element={
            <PrivateRoute><Layout><Dashboard /></Layout></PrivateRoute>
          } />
          <Route path="/builder" element={
            <PrivateRoute><Layout><Builder /></Layout></PrivateRoute>
          } />
          <Route path="/document/:type/:id" element={
            <PrivateRoute><Layout><DocumentView /></Layout></PrivateRoute>
          } />
        </Routes>
      </BrowserRouter>
    </AuthProvider>
  );
}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

import { useEffect, useState } from 'react';
import { useParams, useNavigate, Link } from 'react-router-dom';
import { apiFetch, downloadPDF } from '../api';

export default function DocumentView() {
  const { type, id } = useParams();
  const navigate     = useNavigate();
  const [doc, setDoc]       = useState(null);
  const [loading, setLoading] = useState(true);
  const [downloading, setDownloading] = useState(false);

  const isResume = type === 'resume';

  useEffect(() => {
    const path = isResume ? `/resumes/${id}` : `/resumes/cover-letter/${id}`;
    apiFetch(path).then(setDoc).finally(() => setLoading(false));
  }, [id, type]);

  async function handleDownload() {
    setDownloading(true);
    try {
      const pdfPath = isResume ? `/pdf/resume/${id}` : `/pdf/cover-letter/${id}`;
      const filename = isResume
        ? `${(doc.full_name || 'resume').replace(/\s+/g, '_')}_resume.pdf`
        : `cover_letter_${id.slice(0, 8)}.pdf`;
      await downloadPDF(pdfPath, filename);
    } catch {
      alert('PDF download failed. Please try again.');
    } finally {
      setDownloading(false);
    }
  }

  async function handleDelete() {
    if (!confirm('Delete this document?')) return;
    const path = isResume ? `/resumes/${id}` : `/resumes/cover-letter/${id}`;
    await apiFetch(path, { method: 'DELETE' });
    navigate('/');
  }

  if (loading) return (
    <div className="flex items-center justify-center py-24">
      <div className="w-8 h-8 border-4 border-brand-500 border-t-transparent rounded-full animate-spin" />
    </div>
  );

  if (!doc) return (
    <div className="text-center py-24">
      <p className="text-slate-500">Document not found.</p>
      <Link to="/" className="text-brand-600 hover:underline text-sm mt-2 inline-block">← Back to dashboard</Link>
    </div>
  );

  // Render AI content with basic section formatting
  const renderContent = (content) => {
    if (!content) return <p className="text-slate-400">No content generated.</p>;

    return content.split('\n').map((line, i) => {
      if (line.startsWith('## ')) {
        return (
          <h3 key={i} className="text-xs font-bold text-brand-600 uppercase tracking-widest mt-6 mb-2 pb-1 border-b border-brand-100">
            {line.replace('## ', '')}
          </h3>
        );
      }
      if (line.startsWith('- ') || line.startsWith('• ')) {
        return (
          <li key={i} className="text-slate-700 text-sm leading-relaxed ml-4 list-disc">
            {line.replace(/^[-•]\s/, '')}
          </li>
        );
      }
      if (line.trim() === '') return <div key={i} className="h-2" />;
      return <p key={i} className="text-slate-700 text-sm leading-relaxed">{line}</p>;
    });
  };

  return (
    <div className="max-w-3xl mx-auto space-y-6">

      {/* Top bar */}
      <div className="flex items-start justify-between gap-4">
        <div>
          <Link to="/" className="text-sm text-slate-400 hover:text-slate-600 mb-2 inline-block">← Dashboard</Link>
          <h1 className="text-xl font-bold text-slate-900">{doc.title}</h1>
          <p className="text-sm text-slate-400 mt-0.5">
            Created {new Date(doc.created_at).toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' })}
          </p>
        </div>
        <div className="flex gap-2 shrink-0">
          <button
            onClick={handleDownload}
            disabled={downloading}
            className="btn-primary text-sm py-2 flex items-center gap-2"
          >
            {downloading ? (
              <>
                <div className="w-3 h-3 border-2 border-white border-t-transparent rounded-full animate-spin" />
                Exporting...
              </>
            ) : '⬇ Download PDF'}
          </button>
          <button onClick={handleDelete} className="btn-outline text-sm text-red-500 border-red-200 hover:bg-red-50 py-2">
            Delete
          </button>
        </div>
      </div>

      {/* Info chips */}
      {isResume && (
        <div className="flex flex-wrap gap-2">
          {doc.full_name && <span className="text-xs bg-slate-100 text-slate-600 px-3 py-1 rounded-full font-medium">{doc.full_name}</span>}
          {doc.job_title && <span className="text-xs bg-brand-50 text-brand-600 px-3 py-1 rounded-full font-medium">{doc.job_title}</span>}
          {doc.location  && <span className="text-xs bg-slate-100 text-slate-600 px-3 py-1 rounded-full font-medium">{doc.location}</span>}
        </div>
      )}

      {/* Document content */}
      <div className="card bg-white shadow-sm">
        <div className="prose max-w-none">
          {renderContent(doc.content)}
        </div>
      </div>

      {/* Tip */}
      <p className="text-xs text-slate-400 text-center pb-4">
        Tip: Download as PDF to submit directly to employers or paste content into LinkedIn.
      </p>
    </div>
  );
}

# ResumeAI — AI Resume & Cover Letter Generator

An AI-powered SaaS app that generates professional,
ATS-friendly resumes and cover letters in seconds.

## Tech Stack
- Frontend: React + Vite + Tailwind CSS
- Backend:  Node.js + Express
- Database: Supabase (PostgreSQL + Auth)
- AI:       OpenAI GPT-3.5-turbo
- PDF:      PDFKit

## Setup

### 1. Supabase
- Create a project at supabase.com
- Run supabase/schema.sql in the SQL editor
- Copy Project URL, anon key, and service role key

### 2. OpenAI
- Create an account at platform.openai.com
- Generate an API key (sk-...)

### 3. Backend
  cd backend
  npm install
  cp .env.example .env    ← fill in your values
  npm run dev             ← runs on :4000

### 4. Frontend
  cd frontend
  npm install
  cp .env.example .env    ← fill in your Supabase keys
  npm run dev             ← runs on :5173

## Features
- Email/password auth via Supabase
- AI resume generation (GPT-3.5-turbo)
- AI cover letter generation
- PDF export for both document types
- Daily usage limit (5 generations/day on free plan)
- Usage bar with visual indicator
- Save/view/delete documents from dashboard

## Monetization
- Free plan: 5 generations/day
- Paid plan: flip users.plan to 'paid' via Stripe webhook
  to remove the daily cap

## Cost Control
- GPT-3.5-turbo is used (not GPT-4) to minimize cost
- Tokens capped at 1200 (resume) and 600 (cover letter)
- IP + per-user rate limiting on all generation routes
- Generations table logs all usage for monitoring
